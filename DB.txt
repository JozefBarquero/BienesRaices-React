-- 1. BASE DE DATOS
CREATE DATABASE IF NOT EXISTS bienes_raices_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE bienes_raices_db;

CREATE TABLE IF NOT EXISTS personas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    tipo_persona ENUM('fisica', 'juridica') NOT NULL,
    nombre VARCHAR(150) NULL,
    apellidos VARCHAR(150) NULL,
    razon_social VARCHAR(200) NULL,
    documento_identidad VARCHAR(50) NOT NULL UNIQUE,
    tipo_documento ENUM('cedula', 'pasaporte', 'ruc', 'otro') NOT NULL,
    email VARCHAR(150) NULL,
    telefono VARCHAR(30) NULL,
    direccion VARCHAR(255) NULL,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 2. USUARIOS CON ROLES
CREATE TABLE IF NOT EXISTS trabajadores (
    id VARCHAR(30) PRIMARY KEY,
    persona_id INT NOT NULL UNIQUE,
    email VARCHAR(150) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NULL,
    rol ENUM('Administrador', 'agente_inmobiliario') NOT NULL DEFAULT 'agente_inmobiliario',
    cargo VARCHAR(100) NULL,
    fecha_ingreso DATE NOT NULL,
    fecha_salida DATE NULL,
    activo BOOLEAN DEFAULT TRUE,
    creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (persona_id) REFERENCES personas(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 3. TIPOS DE INMUEBLE
CREATE TABLE IF NOT EXISTS tipos_inmueble (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS propietarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    persona_id INT NOT NULL UNIQUE,
    notas TEXT NULL,
    activo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (persona_id) REFERENCES personas(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 4. PROPIEDADES
CREATE TABLE IF NOT EXISTS propiedades (
    id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(200) NOT NULL,
    descripcion_corta VARCHAR(300) NOT NULL,
    descripcion_larga TEXT NOT NULL,
    
    -- Ubicación y Geolocalización
    ubicacion VARCHAR(255) NOT NULL,        
    direccion_exacta VARCHAR(255) NOT NULL,    
    latitud DECIMAL(10, 8) NULL,                
    longitud DECIMAL(11, 8) NULL,                
    
    -- Tipo y Operación
    tipo_inmueble_id INT NULL,
    tipo_operacion ENUM('venta', 'alquiler', 'ambos') NOT NULL DEFAULT 'venta',
    
    -- Precios y Divisa
    precio DECIMAL(12, 2) NOT NULL,
    moneda ENUM('USD', 'CRC', 'EUR') NOT NULL DEFAULT 'USD',
    tipo_precio ENUM('fijo', 'negociable') NOT NULL DEFAULT 'fijo',
    
    -- Características Físicas
    habitaciones TINYINT UNSIGNED NULL,
    banos DECIMAL(3,1) UNSIGNED NULL,          
    estacionamientos TINYINT UNSIGNED DEFAULT 0,
    area_construida DECIMAL(10,2) UNSIGNED NULL, 
    area_terreno DECIMAL(10,2) UNSIGNED NULL,    
    
    -- Contenido Multimedia
    imagen_principal VARCHAR(500) NOT NULL,
    url_video VARCHAR(500) NULL,                
    
    -- Control, Visibilidad, Creador y Estado
    destacada BOOLEAN DEFAULT FALSE,          
    activa BOOLEAN DEFAULT TRUE,                
    creado_por VARCHAR(30) NULL,
    propietario_id INT NULL,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    -- Claves Foráneas
    FOREIGN KEY (tipo_inmueble_id) REFERENCES tipos_inmueble(id) ON DELETE SET NULL,
    FOREIGN KEY (creado_por) REFERENCES trabajadores(id) ON DELETE SET NULL,
    FOREIGN KEY (propietario_id) REFERENCES propietarios(id) ON DELETE SET NULL
) ENGINE=InnoDB;

-- 5. GALERÍA DE IMÁGENES
CREATE TABLE IF NOT EXISTS imagenes_propiedad (
    id INT AUTO_INCREMENT PRIMARY KEY,
    propiedad_id INT NOT NULL,
    url_imagen VARCHAR(500) NOT NULL,
    orden INT DEFAULT 0,                        
    FOREIGN KEY (propiedad_id) REFERENCES propiedades(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 6. MENSAJES DE CONTACTO
CREATE TABLE IF NOT EXISTS mensajes_contacto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    propiedad_id INT NULL,
    nombre_cliente VARCHAR(100) NOT NULL,
    email_cliente VARCHAR(150) NOT NULL,
    telefono_cliente VARCHAR(30) NULL,
    mensaje TEXT NOT NULL,
    estado ENUM('nuevo', 'en_proceso', 'atendido', 'descartado') DEFAULT 'nuevo',
    asignado_a VARCHAR(30) NULL,
    fecha_envio TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_ultimo_contacto TIMESTAMP NULL,
    intentos_contacto INT DEFAULT 0,
    notas_internas TEXT NULL,
    prioridad ENUM('baja', 'media', 'alta') DEFAULT 'media',
    origen VARCHAR(50) NULL,
    activo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (propiedad_id) REFERENCES propiedades(id) ON DELETE SET NULL,
    FOREIGN KEY (asignado_a) REFERENCES trabajadores(id) ON DELETE SET NULL
) ENGINE=InnoDB;

-- 7. ÍNDICES DE RENDIMIENTO
CREATE INDEX idx_propiedades_estado ON propiedades(activa, destacada, tipo_operacion);
CREATE INDEX idx_propiedades_precio ON propiedades(moneda, precio);
CREATE INDEX idx_propiedades_ubicacion ON propiedades(ubicacion);
CREATE INDEX idx_propiedades_tipo ON propiedades(tipo_inmueble_id);
CREATE INDEX idx_propiedades_creador ON propiedades(creado_por);
CREATE INDEX idx_propiedades_propietario ON propiedades(propietario_id);
CREATE INDEX idx_imagenes_orden ON imagenes_propiedad(propiedad_id, orden);
CREATE INDEX idx_mensajes_estado ON mensajes_contacto(estado, activo);
CREATE INDEX idx_mensajes_asignado ON mensajes_contacto(asignado_a);
